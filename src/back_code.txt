import React, { useEffect, useState } from 'react';
import {
  Users, Clock, TrendingUp,
  LogIn, LogOut, RefreshCw, ArrowUpRight
} from 'lucide-react';
import {
  BarChart, Bar, XAxis, YAxis, Tooltip, ResponsiveContainer,
  CartesianGrid, Cell, RadialBarChart, RadialBar, PolarAngleAxis,
  LineChart, Line,
} from 'recharts';
import './Dashboard.css';

/* ── Default state ── */
const DEFAULT = {
  footfall: 0, openingTime: '00:00 AM', closingTime: '00:00 PM',
  busiestDay: 'N/A', busiestHour: 'N/A',
  malePercent: 0, femalePercent: 0, malePeak: 'N/A', femalePeak: 'N/A',
  ageGroups: { '0-20': 0, '21-30': 0, '31-40': 0, '41-50': 0 },
  genderAge: {
    '0-20': { male: 0, female: 0 }, '21-30': { male: 0, female: 0 },
    '31-40': { male: 0, female: 0 }, '41-50': { male: 0, female: 0 },
  },
  entryCount: 0, exitCount: 0,
  maleIn: 0, maleOut: 0, femaleIn: 0, femaleOut: 0,
  hourlyTraffic: {},
};

const AGE_COLORS = ['#22d3ee', '#34d399', '#fbbf24', '#f87171'];

/* ── Components ── */
const ChartTooltip = ({ active, payload, label }) => {
  if (!active || !payload?.length) return null;
  return (
    <div className="dash-tooltip">
      <p className="dash-tooltip__label">{label}</p>
      {payload.map((p, i) => (
        <p key={i} style={{ color: p.fill || p.stroke || p.color }}>
          {p.name}: <strong>{p.value}</strong>
        </p>
      ))}
    </div>
  );
};

const StatCard = ({ icon: Icon, label, value, colorClass, trend }) => (
  <div className={`stat-card ${colorClass}`}>
    <div className="stat-card__icon"><Icon size={20} /></div>
    <div className="stat-card__body">
      <p className="stat-card__label">{label}</p>
      <h2 className="stat-card__value">{value}</h2>
    </div>
    {trend && (
      <span className="stat-card__trend"><ArrowUpRight size={12} /> Live</span>
    )}
  </div>
);

const FlowCard = ({ label, icon: Icon, count, colorClass, subItems }) => (
  <div className={`flow-card ${colorClass}`}>
    <div className="flow-card__head">
      <div className="flow-card__icon"><Icon size={18} /></div>
      <div>
        <p className="flow-card__label">{label}</p>
        <h2 className="flow-card__count">{count.toLocaleString()}</h2>
      </div>
    </div>
    <div className="flow-card__sub">
      {subItems.map((s, i) => (
        <div key={i} className="flow-card__sub-row">
          <span className="flow-card__sub-dot" style={{ background: s.color }} />
          <span className="flow-card__sub-label">{s.label}</span>
          <span className="flow-card__sub-val" style={{ color: s.color }}>{s.value}</span>
        </div>
      ))}
    </div>
  </div>
);

const GaugeChart = ({ percent, color, label }) => (
  <div className="gauge">
    <div className="gauge__svg-wrap">
      <ResponsiveContainer width={112} height={112}>
        <RadialBarChart
          cx="50%" cy="50%"
          innerRadius="64%" outerRadius="90%"
          startAngle={210} endAngle={-30}
          data={[{ value: percent, fill: color }]}
        >
          <PolarAngleAxis type="number" domain={[0, 100]} tick={false} />
          <RadialBar
            dataKey="value"
            cornerRadius={6}
            background={{ fill: 'rgba(255,255,255,0.05)' }}
          />
        </RadialBarChart>
      </ResponsiveContainer>
      <div className="gauge__center">
        <span className="gauge__pct" style={{ color }}>{percent}%</span>
        <span className="gauge__label">{label}</span>
      </div>
    </div>
  </div>
);

/* ── Main Dashboard ── */
export default function Dashboard() {
  const [data, setData] = useState(DEFAULT);
  const [loading, setLoading] = useState(false);
  const [lastUpdated, setLast] = useState(null);
  
  // Date states
  const todayStr = new Date().toISOString().split('T')[0];
  const [dateInput, setDateInput] = useState(todayStr); // For the input field
  const [activeDate, setActiveDate] = useState(todayStr); // For the API query

  const fetchData = async () => {
    setLoading(true);
    try {
      const res = await fetch(`http://localhost:5000/api/dashboard_stats?date=${activeDate}`);
      const json = await res.json();
      setData(json);
      setLast(new Date().toLocaleTimeString());
    } catch (err) {
      console.error("Fetch error:", err);
    }
    setLoading(false);
  };

  // Sync data when activeDate changes
  useEffect(() => {
    fetchData();
    
    // Live polling only if viewing today's data
    let intervalId;
    if (activeDate === todayStr) {
      intervalId = setInterval(fetchData, 5000);
    }
    return () => clearInterval(intervalId);
  }, [activeDate]);

  const handleApplyDate = () => {
    setActiveDate(dateInput);
  };

  /* ── Data Formatting ── */
  const ageData = Object.entries(data.ageGroups).map(([k, v]) => ({ name: k, value: v }));
  const genderAgeData = Object.entries(data.genderAge).map(([k, v]) => ({
    name: k, male: v.male, female: v.female,
  }));

  const hourlyData = Object.entries(data.hourlyTraffic || {})
    .map(([hour, counts]) => ({
      hour,
      entry: counts.entry ?? 0,
      exit: counts.exit ?? 0,
      _sort: (() => {
        const parts = hour.trim().split(' ');
        if (parts.length < 2) return 0;
        const [h, period] = parts;
        let n = parseInt(h, 10);
        if (period === 'PM' && n !== 12) n += 12;
        if (period === 'AM' && n === 12) n = 0;
        return n;
      })(),
    }))
    .sort((a, b) => a._sort - b._sort);

  const occupancy = Math.max(0, data.entryCount - data.exitCount);
  const occupancyPct = data.entryCount > 0
    ? Math.min(100, (occupancy / data.entryCount) * 100).toFixed(1)
    : 0;

  return (
    <div className="dash-page">
      {/* ── Topbar ── */}
      <div className="dash-topbar">
        <div>
          <h1 className="dash-title">Analytics Dashboard</h1>
          <p className="dash-sub">
            Showing data for: <strong>{activeDate}</strong>
            <span style={{ margin: '0 8px', opacity: 0.3 }}>|</span>
            Last updated: {lastUpdated ?? '—'}
            {loading && <span className="spinner" style={{ marginLeft: 8 }} />}
          </p>
        </div>

        <div className="dash-controls">
          <div className='date-range-wrap'><input 
            type="date" 
            className="date-input"
            value={dateInput}
            onChange={(e) => setDateInput(e.target.value)}
          /></div>
          <button className="btn-apply" onClick={handleApplyDate}>
            Apply
          </button>
          <button className="btn-refresh" onClick={fetchData}>
            <RefreshCw size={14} /> Refresh
          </button>
        </div>
      </div>

      {/* ── KPI grid ── */}
      <div className="dash-kpi-grid">
        <StatCard icon={Users} label="Total Footfall" value={data.footfall.toLocaleString()} colorClass="ci" trend={activeDate === todayStr} />
        <StatCard icon={Clock} label="Opening Time" value={data.openingTime} colorClass="ce" />
        <StatCard icon={Clock} label="Closing Time" value={data.closingTime} colorClass="cr" />
        <StatCard icon={TrendingUp} label="Busiest Day" value={data.busiestDay} colorClass="cv" />
        <StatCard icon={TrendingUp} label="Busiest Hour" value={data.busiestHour} colorClass="ca" />
        <StatCard icon={Users} label="Male Peak Hour" value={data.malePeak} colorClass="cc" />
        <StatCard icon={Users} label="Female Peak Hour" value={data.femalePeak} colorClass="cp" />
      </div>

      {/* ── Flow row ── */}
      <div className="dash-flow-row">
        <FlowCard
          icon={LogIn} label="Total Entries" count={data.entryCount}
          colorClass="flow-entry"
          subItems={[
            { label: 'Male in', value: data.maleIn, color: '#818cf8' },
            { label: 'Female in', value: data.femaleIn, color: '#f472b6' },
          ]}
        />
        <FlowCard
          icon={LogOut} label="Total Exits" count={data.exitCount}
          colorClass="flow-exit"
          subItems={[
            { label: 'Male out', value: data.maleOut, color: '#818cf8' },
            { label: 'Female out', value: data.femaleOut, color: '#f472b6' },
          ]}
        />
        <div className="occupancy-card">
          <p className="occupancy-card__label">Current Occupancy</p>
          <h2 className="occupancy-card__count">{occupancy.toLocaleString()}</h2>
          <p className="occupancy-card__sub">people currently inside</p>
          <div className="occ-bar">
            <div className="occ-bar__fill" style={{ width: `${occupancyPct}%` }} />
          </div>
          <p className="occ-pct">{occupancyPct}% of peak entries still inside</p>
        </div>
        <div className="gauge-card">
          <p className="gauge-card__label">Gender Split</p>
          <div className="gauge-card__row">
            <GaugeChart percent={data.malePercent} color="#818cf8" label="Male" />
            <GaugeChart percent={data.femalePercent} color="#f472b6" label="Female" />
          </div>
        </div>
      </div>

      {/* ── Gender banner ── */}
      <div className="gender-banner">
        <div className="gender-banner__side gender-banner__side--male">
          <span className="gender-banner__sym">♂</span>
          <div>
            <p>Male visitors</p>
            <h3>{data.malePercent}%</h3>
          </div>
        </div>
        <div className="gender-banner__track">
          <div className="gender-banner__fill" style={{ width: `${data.malePercent}%` }} />
        </div>
        <div className="gender-banner__side gender-banner__side--female">
          <div className="gender-banner__right">
            <p>Female visitors</p>
            <h3>{data.femalePercent}%</h3>
          </div>
          <span className="gender-banner__sym">♀</span>
        </div>
      </div>

      {/* ── Charts ── */}
      <div className="dash-charts">
        <div className="chart-card">
          <div className="chart-card__head">
            <h3 className="chart-card__title">Age Groups</h3>
            <span className="badge badge--emerald">Count</span>
          </div>
          <div className="legend-row">
            {ageData.map((d, i) => (
              <span key={i} className="legend-item">
                <span className="legend-dot" style={{ background: AGE_COLORS[i % 4] }} />
                {d.name}
              </span>
            ))}
          </div>
          <ResponsiveContainer width="100%" height={240}>
            <BarChart data={ageData} barSize={42}>
              <CartesianGrid strokeDasharray="3 3" stroke="rgba(255,255,255,0.055)" />
              <XAxis dataKey="name" tick={{ fill: '#94a3b8', fontSize: 12 }} axisLine={false} tickLine={false} />
              <YAxis tick={{ fill: '#94a3b8', fontSize: 11 }} axisLine={false} tickLine={false} />
              <Tooltip content={<ChartTooltip />} />
              <Bar dataKey="value" radius={[6, 6, 0, 0]}>
                {ageData.map((_, i) => <Cell key={i} fill={AGE_COLORS[i % 4]} />)}
              </Bar>
            </BarChart>
          </ResponsiveContainer>
        </div>

        <div className="chart-card">
          <div className="chart-card__head">
            <h3 className="chart-card__title">Hourly Traffic</h3>
            <div style={{ display: 'flex', gap: 6 }}>
              <span className="badge badge--emerald">In {data.entryCount}</span>
              <span className="badge badge--rose">Out {data.exitCount}</span>
            </div>
          </div>
          <div className="legend-row">
            <span className="legend-item"><span className="legend-dot" style={{ background: '#34d399' }} />Entry</span>
            <span className="legend-item"><span className="legend-dot" style={{ background: '#f87171' }} />Exit</span>
          </div>
          <ResponsiveContainer width="100%" height={240}>
            <LineChart data={hourlyData} margin={{ top: 4, right: 8, left: 0, bottom: 4 }}>
              <CartesianGrid strokeDasharray="3 3" stroke="rgba(255,255,255,0.055)" />
              <XAxis dataKey="hour" tick={{ fill: '#94a3b8', fontSize: 11 }} axisLine={false} tickLine={false} />
              <YAxis tick={{ fill: '#94a3b8', fontSize: 11 }} axisLine={false} tickLine={false} allowDecimals={false} />
              <Tooltip content={<ChartTooltip />} />
              <Line type="monotone" dataKey="entry" name="Entry" stroke="#34d399" strokeWidth={2.5} dot={{ r: 3, fill: '#34d399' }} />
              <Line type="monotone" dataKey="exit" name="Exit" stroke="#f87171" strokeWidth={2.5} dot={{ r: 3, fill: '#f87171' }} />
            </LineChart>
          </ResponsiveContainer>
        </div>

        <div className="chart-card chart-card--full">
          <div className="chart-card__head">
            <h3 className="chart-card__title">Gender vs Age Breakdown</h3>
            <div style={{ display: 'flex', gap: 6 }}>
              <span className="badge badge--indigo">Male: {genderAgeData.reduce((s, d) => s + d.male, 0)}</span>
              <span className="badge badge--pink">Female: {genderAgeData.reduce((s, d) => s + d.female, 0)}</span>
            </div>
          </div>
          <div className="legend-row">
            <span className="legend-item"><span className="legend-dot" style={{ background: '#6366f1' }} />Male</span>
            <span className="legend-item"><span className="legend-dot" style={{ background: '#f472b6' }} />Female</span>
          </div>
          <ResponsiveContainer width="100%" height={280}>
            <BarChart data={genderAgeData} barGap={4} barSize={26}>
              <CartesianGrid strokeDasharray="3 3" stroke="rgba(255,255,255,0.055)" />
              <XAxis dataKey="name" tick={{ fill: '#94a3b8', fontSize: 12 }} axisLine={false} tickLine={false} />
              <YAxis tick={{ fill: '#94a3b8', fontSize: 11 }} axisLine={false} tickLine={false} />
              <Tooltip content={<ChartTooltip />} />
              <Bar dataKey="male" fill="#6366f1" radius={[5, 5, 0, 0]} name="Male" />
              <Bar dataKey="female" fill="#f472b6" radius={[5, 5, 0, 0]} name="Female" />
            </BarChart>
          </ResponsiveContainer>
        </div>
      </div>
    </div>
  );
}
